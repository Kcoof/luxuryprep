+function mapRow(row: Record<string, unknown>): ClosingRecord {
+  return {
+    id: String(row.id ?? ''),
+    branchId: String(row.branch_id ?? ''),
+    branchName: String(row.branch_name ?? ''),
+    businessDate: String(row.business_date ?? ''),
+    fields: (row.fields as ClosingRecord['fields']) ?? {

+    },
+    shortageOrExcess: Number(row.shortage_or_excess ?? 0),

+    createdAt: String(row.created_at ?? new Date().toISOString()),
+  };

+export async function findExistingClosing(
+  branchId: string,
+  businessDate: string
+): Promise<ClosingRecord | null> {
+  try {
+    const { getSupabase } = await import('./supabase');
+    const supabase = getSupabase();
+    if (supabase) {
+      const { data } = await supabase
+        .from('daily_closings')
+        .select('*')
+        .eq('branch_id', branchId)
+        .eq('business_date', businessDate)
+        .maybeSingle();
+      if (data) return mapRow(data as Record<string, unknown>);
+    }
+  } catch (e) {
+    console.warn('findExistingClosing Supabase failed, using local', e);
+  }
+  const all = readLocal();
+  return all.find((c) => c.branchId === branchId && c.businessDate === businessDate) ?? null;
+}

+export async function saveClosing(rec: ClosingRecord): Promise<void> {
+  try {
+    const { getSupabase } = await import('./supabase');
+    const supabase = getSupabase();
+    if (supabase) {
+      const { error } = await supabase.from('daily_closings').upsert({
+        id: rec.id,
+        branch_id: rec.branchId,
+        branch_name: rec.branchName,
+        business_date: rec.businessDate,
+        fields: rec.fields,
+        shortage_or_excess: rec.shortageOrExcess,
+        status: rec.status,
+        manual_actual_cash: rec.manualActualCash ?? null,
+        images: rec.images ?? null,
+        created_at: rec.createdAt,
+      });
+      if (!error) return;
+      console.warn('saveClosing Supabase error', error);
+    }
+  } catch (e) {
+    console.warn('saveClosing fallback to local', e);
+  }
+  const all = readLocal();
+  const idx = all.findIndex(
+    (c) => c.branchId === rec.branchId && c.businessDate === rec.businessDate
+  );
+  if (idx >= 0) all[idx] = rec;
+  else all.push(rec);
+  writeLocal(all);

+-- 002_daily_closings.sql
+-- M2 Cashier flow: store daily closing submissions.
+-- Storage bucket for proof images: closing-images (create via Supabase Studio if not present).
+
+create table if not exists public.daily_closings (
+  id text primary key,
+  branch_id text not null references public.branches(id) on delete cascade,
+  branch_name text not null,
+  business_date date not null,
+  fields jsonb not null default '{}'::jsonb,
+  shortage_or_excess numeric not null default 0,
+  status text not null default 'pending' check (status in ('pending','approved','rejected')),
+  manual_actual_cash numeric,
+  images jsonb,
+  created_at timestamptz not null default now()
+);
+
+create index if not exists daily_closings_branch_date_idx
+  on public.daily_closings (branch_id, business_date);
+
+alter table public.daily_closings enable row level security;
