+"use client";

+import seedBranches from "../../supabase/seed_branches.json";
+import type { Branch } from "../types";
+import { getSupabase, isSupabaseConfigured } from "./supabase";
+
+export function getSeedBranches(): Branch[] {
+  return (seedBranches as Branch[]).map((b) => ({
+    id: b.id,
+    name: b.name,
+    city: b.city ?? "",
+  }));
+}
+
+export async function listBranches(): Promise<Branch[]> {
+  if (!isSupabaseConfigured) {
+    return getSeedBranches();
+  }
+  try {
+    const supabase = getSupabase();
+    const { data, error } = await supabase
+      .from("branches")
+      .select("id, name, city")
+      .order("id", { ascending: true });
+    if (error || !data || data.length === 0) {
+      return getSeedBranches();
+    }
+    return data as Branch[];
+  } catch {
+    return getSeedBranches();
+  }
+}
+
+export function findBranch(branches: Branch[], id?: string): Branch | undefined {
+  if (!id) return undefined;
+  return branches.find((b) => b.id === id);

+import type {
+  AuditAction,
+  DailyClosing,
+  DailyClosingAuditLog,
+  FinancialFields,
+  UserRole,
+} from "../types";
+import { computeShortageOrExcess } from "../types";
+import { getSupabase, isSupabaseConfigured } from "./supabase";
+
+export interface NewClosingInput {
+  branchId: string;
+  businessDate: string;
+  reviewedData: FinancialFields;
+  zReportImageUrl?: string;
+  paymentProofImageUrls?: string[];
+  manualActualCash?: number;
+}
+
+export async function createClosing(
+  input: NewClosingInput,
+): Promise<{ ok: boolean; closing?: DailyClosing; error?: string }> {
+  const payload = buildClosingPayload(input);
+  if (!isSupabaseConfigured) {
+    const closing = makeLocalClosing(payload);
+    return { ok: true, closing };
+  }
+  try {
+    const supabase = getSupabase();
+    const { data, error } = await supabase
+      .from("daily_closings")
+      .insert(payload)
+      .select()
+      .single();
+    if (error) return { ok: false, error: error.message };
+    await appendAuditLog(supabase, data.id, "cashier", undefined, "cashier_confirmed");
+    return { ok: true, closing: data as DailyClosing };
+  } catch (e) {
+    return { ok: false, error: (e as Error).message };
+  }
+}
+
+function buildClosingPayload(input: NewClosingInput) {
+  const shortage = computeShortageOrExcess(
+    input.reviewedData.cashActualHanded,
+    input.reviewedData.cashSystem,
+  );
+  return {
+    branch_id: input.branchId,
+    business_date: input.businessDate,
+    status: "pending" as const,
+    z_report_image_url: input.zReportImageUrl ?? null,
+    payment_proof_image_urls: input.paymentProofImageUrls ?? [],
+    reviewed_data: { ...input.reviewedData, shortageOrExcess: shortage },
+    manual_actual_cash: input.manualActualCash ?? null,
+  };
+}
+
+function makeLocalClosing(payload: ReturnType<typeof buildClosingPayload>): DailyClosing {
+  const now = new Date().toISOString();
+  return {
+    id: `local_${Date.now()}`,
+    branchId: payload.branch_id,
+    businessDate: payload.business_date,
+    status: "pending",
+    zReportImageUrl: payload.z_report_image_url ?? undefined,
+    paymentProofImageUrls: payload.payment_proof_image_urls ?? [],
+    reviewedData: payload.reviewed_data,
+    manualActualCash: payload.manual_actual_cash ?? undefined,
+    createdAt: now,
+    updatedAt: now,
+  };
+}
+
+async function appendAuditLog(
+  supabase: ReturnType<typeof getSupabase>,
+  closingId: string,
+  role: UserRole,
+  actorId: string | undefined,
+  action: AuditAction,
+) {
+  await supabase.from("daily_closing_audit_logs").insert({
+    closing_id: closingId,
+    actor_role: role,
+    actor_id: actorId ?? null,
+    action,
+  });
+}
